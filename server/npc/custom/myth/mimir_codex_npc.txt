//=============================================================================
// Myth of Thor - Guardião do Códice de Mímir (Bestiário e Relíquias)
//=============================================================================
prontera,156,185,5	script	Mimir Guardian#myth	4_M_SAGE_A,{
	mes "[Mimir]";
	mes "I keep the memories of the nine realms beneath the crown of Yggdrasil.";
	mes "Your journeys and battles are recorded in this great tome.";
	next;
	switch(select("View Accumulated Knowledge:Claim Milestone Rewards:How does the Codex work?:Cancel")) {
	case 1:
		mes "[Mimir]";
		mes "Mimir's Knowledge Points: ^0000FF" + MIMIR_PTS + "^000000";
		mes "Keep hunting creatures and exploring dungeons to expand your records.";
		close;

	case 2:
		mes "[Mimir]";
		mes "Checking your knowledge milestones...";
		if (MIMIR_PTS >= 10 && !(MIMIR_REWARD & 1)) {
			MIMIR_REWARD = MIMIR_REWARD | 1;
			mes "Congratulations! You reached the 10 Point Milestone: [Midgard Observer]!";
			specialeffect2 EF_LEVELUP;
			close;
		}
		if (MIMIR_PTS >= 30 && !(MIMIR_REWARD & 2)) {
			MIMIR_REWARD = MIMIR_REWARD | 2;
			mes "Congratulations! You reached the 30 Point Milestone: [Prontera Scholar]!";
			specialeffect2 EF_LEVELUP;
			close;
		}
		mes "No new milestone reached at the moment.";
		close;

	case 3:
		mes "[Mimir]";
		mes "Each monster species has 3 mastery levels:";
		mes "- Level 1 (1 kill): Basic identification.";
		mes "- Level 2 (50 kills): HP shown on the target bar.";
		mes "- Level 3 (250 kills): Drop table and a permanent +1% damage against the species.";
		close;

	case 4:
		mes "[Mimir]";
		mes "Wisdom is the greatest of all weapons.";
		close;
	}
}
