prontera,152,185,5	script	Synthesis Smith#myth	4_M_BLACKSMITH,{
	mes "[Hephaestus]";
	mes "Greetings, warrior. I am the guardian of the aether forges.";
	mes "I can imbue or reforge mystic attributes on your equipment.";
	next;
	switch(select("What is Synthesis?:Imbue Mystic Attributes:Reforge Attributes (Reroll):Cancel")) {
	case 1:
		mes "[Hephaestus]";
		mes "Synthesis lets your weapons and armor gain ^0000FF1 to 3 additional random attributes^000000.";
		mes "The attributes depend on the equipment type and are never lost when refining.";
		close;

	case 2:
		mes "[Hephaestus]";
		mes "To imbue a clean weapon or armor, I need:";
		mes "- 1x Oridecon or Elunium";
		mes "- 2x Synthesis Fragments";
		mes "- 50,000 Zeny";
		next;
		if (Zeny < 50000) {
			mes "[Hephaestus]";
			mes "You do not have enough Zeny.";
			close;
		}
		if (countitem(984) < 1 && countitem(985) < 1) { // Oridecon ou Elunium
			mes "[Hephaestus]";
			mes "You need at least 1 Oridecon or 1 Elunium.";
			close;
		}
		mes "[Hephaestus]";
		mes "Select the equipment you want to imbue from your inventory:";
		close;

	case 3:
		mes "[Hephaestus]";
		mes "Reforging rerolls new attributes while keeping cards and refines.";
		mes "Cost: 1x Oridecon/Elunium, 3x Fragments and 75,000 Zeny.";
		close;

	case 4:
		mes "[Hephaestus]";
		mes "May Thor's flame guide your steps.";
		close;
	}
}
