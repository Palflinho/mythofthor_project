//===== Myth of Thor =========================================
//= Server messages in English (default rAthena language pack).
//= Accounts that previously used another language (saved #langtype)
//= are switched back to English on login.
//============================================================
-	script	MythIdioma	-1,{
OnPCLoginEvent:
	if (#langtype != 0)
		atcommand "@langtype eng";
	end;
}
